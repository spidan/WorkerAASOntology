PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
PREFIX owl: <http://www.w3.org/2002/07/owl#>
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>
PREFIX tools: <http://example.org/dfki/twinmap/Tools/>
PREFIX cap: <http://example.org/dfki/twinmap/Capabilities/>
PREFIX toy: <http://example.org/dfki/twinmap/toyCar/>

SELECT DISTINCT ?screw ?worker
	WHERE { ?screw a tools:HexScrew, ?threadType. 
	                 ?worker cap:providesService ?service.
	                 FILTER (?service =  cap:AssembleToyCar )
	                 ?service cap:hasParameter ?truck.
	                  ?truck toy:hasWheel/toy:attachedAt ?attachment. ?attachment a ?threadType . ?threadType rdfs:subClassOf tools:Thread.  }
